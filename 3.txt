name: Build Sovereign Shield APK

on:
  push:
    branches: [ main ]
  workflow_dispatch: # يسمح لك بتشغيل البناء يدوياً

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout code
        uses: actions/checkout@v3

      - name: Set up Python
        uses: actions/setup-python@v4
        with:
          python-version: '3.9'

      - name: Install dependencies
        run: |
          sudo apt update
          sudo apt install -y git zip unzip openjdk-17-jdk python3-pip autoconf libtool pkg-config zlib1g-dev libncurses5-dev libncursesw5-dev libtinfo5 cmake libffi-dev libssl-dev
          pip3 install --user --upgrade Cython==0.29.33 buildozer

      - name: Build with Buildozer
        run: |
          # إضافة مسار buildozer إلى النظام
          export PATH=$PATH:~/.local/bin
          # بدء عملية البناء (سيقوم بتحميل NDK و SDK تلقائياً)
          buildozer android debug
        env:
          ACCEPT_SDK_LICENSE: "yes"

      - name: Upload APK Artifact
        uses: actions/upload-artifact@v3
        with:
          name: SovereignShield-APK
          path: bin/*.apk
